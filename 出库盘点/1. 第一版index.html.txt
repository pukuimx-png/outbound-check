    // ============================================================
    //  核心工具：根据列头名称查找列索引（不依赖固定索引）
    // ============================================================
    function findColumnIndexByHeader(headers, targetNames) {
        if (!Array.isArray(headers)) return -1;
        const lowerHeaders = headers.map(h => String(h).trim().toLowerCase());
        const targets = Array.isArray(targetNames) ? targetNames : [targetNames];
        const lowerTargets = targets.map(t => String(t).trim().toLowerCase());
        for (let i = 0; i < lowerHeaders.length; i++) {
            if (lowerTargets.some(t => lowerHeaders[i].includes(t) || t.includes(lowerHeaders[i]))) {
                return i;
            }
        }
        return -1;
    }

    // ============================================================
    //  解析 Excel 文件
    // ============================================================
    function parseExcelFile(file) {
        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.onload = (e) => {
                try {
                    const data = new Uint8Array(e.target.result);
                    const workbook = XLSX.read(data, { type: 'array', cellDates: true });
                    const firstSheet = workbook.Sheets[workbook.SheetNames[0]];
                    const json = XLSX.utils.sheet_to_json(firstSheet, { defval: '' });
                    resolve(json);
                } catch (err) { reject(err); }
            };
            reader.onerror = reject;
            reader.readAsArrayBuffer(file);
        });
    }

    // ============================================================
    //  主逻辑
    // ============================================================
    let file1Data = null;
    let file2Data = null;
    let file1Name = '';
    let file2Name = '';
    // 存储当前导出的数据（用于下载）
    let currentExportData = null;
    let currentExportHeaders = [];
    let currentDateValue = '';

    const fileInput1 = document.getElementById('fileInput1');
    const fileInput2 = document.getElementById('fileInput2');
    const status1 = document.getElementById('status1');
    const status2 = document.getElementById('status2');
    const processBtn = document.getElementById('processBtn');
    const resetBtn = document.getElementById('resetBtn');
    const resultArea = document.getElementById('resultArea');
    const infoMsg = document.getElementById('infoMsg');
    const downloadBtn = document.getElementById('downloadBtn');

    const tableHead = document.getElementById('tableHead');
    const tableBody = document.getElementById('tableBody');
    const extractedDate = document.getElementById('extractedDate');
    const totalBoxes = document.getElementById('totalBoxes');
    const totalBoxesStock = document.getElementById('totalBoxesStock');
    const checkResult = document.getElementById('checkResult');
    const colPosition = document.getElementById('colPosition');

    // 监听文件上传
    fileInput1.addEventListener('change', async function(e) {
        const file = e.target.files[0];
        if (!file) { status1.textContent = '未上传'; file1Data = null; return; }
        status1.textContent = '⏳ 读取中...';
        try {
            file1Data = await parseExcelFile(file);
            file1Name = file.name;
            status1.textContent = `✅ ${file.name} (${file1Data.length} 行)`;
        } catch (err) {
            status1.textContent = '❌ 解析失败，请检查文件格式';
            file1Data = null;
            console.error(err);
        }
    });

    fileInput2.addEventListener('change', async function(e) {
        const file = e.target.files[0];
        if (!file) { status2.textContent = '未上传'; file2Data = null; return; }
        status2.textContent = '⏳ 读取中...';
        try {
            file2Data = await parseExcelFile(file);
            file2Name = file.name;
            status2.textContent = `✅ ${file.name} (${file2Data.length} 行)`;
        } catch (err) {
            status2.textContent = '❌ 解析失败，请检查文件格式';
            file2Data = null;
            console.error(err);
        }
    });

    // 重置
    resetBtn.addEventListener('click', function() {
        fileInput1.value = '';
        fileInput2.value = '';
        file1Data = null;
        file2Data = null;
        status1.textContent = '未上传';
        status2.textContent = '未上传';
        resultArea.style.display = 'none';
        infoMsg.textContent = '请上传两个文件后点击生成';
        tableHead.innerHTML = '';
        tableBody.innerHTML = '';
        currentExportData = null;
        currentExportHeaders = [];
        currentDateValue = '';
    });

    // 生成拣货单
    processBtn.addEventListener('click', function() {
        if (!file1Data || !file2Data) {
            infoMsg.textContent = '⚠️ 请上传两个文件（出库单和库存表）';
            return;
        }

        // 获取列头（第一行对象的 keys）
        const headers1 = Object.keys(file1Data[0] || {});
        const headers2 = Object.keys(file2Data[0] || {});

        // 动态识别箱号列
        const boxIndex1 = findColumnIndexByHeader(headers1, ['箱号', 'BOX', 'Caja', 'Box', '箱']);
        const boxIndex2 = findColumnIndexByHeader(headers2, ['箱号', 'BOX', 'Caja', 'Box', '箱']);

        if (boxIndex1 === -1 || boxIndex2 === -1) {
            infoMsg.textContent = '❌ 未找到"箱号"列，请检查表头命名（箱号/BOX/Caja）';
            return;
        }

        // 提取箱号数据
        const boxKey1 = headers1[boxIndex1];
        const boxKey2 = headers2[boxIndex2];
        const boxes1 = file1Data.map(row => String(row[boxKey1] || '').trim()).filter(v => v !== '');
        const boxes2 = file2Data.map(row => String(row[boxKey2] || '').trim()).filter(v => v !== '');

        // 提取业务日期（优先从出库单找 "日期"/"Fecha"/"Date"）
        const dateIndex = findColumnIndexByHeader(headers1, ['日期', 'Fecha', 'Date', 'fecha', 'date']);
        let dateValue = '—';
        if (dateIndex !== -1) {
            const key = headers1[dateIndex];
            const firstVal = file1Data[0]?.[key];
            if (firstVal) {
                if (typeof firstVal === 'number') {
                    const d = new Date((firstVal - 25569) * 86400 * 1000);
                    dateValue = d.toISOString().split('T')[0];
                } else if (firstVal instanceof Date) {
                    dateValue = firstVal.toISOString().split('T')[0];
                } else {
                    dateValue = String(firstVal).split(' ')[0];
                }
            }
        }
        currentDateValue = dateValue;

        // 存储当前数据以便导出
        currentExportData = file1Data;
        currentExportHeaders = headers1;

        // 显示结果
        resultArea.style.display = 'flex';
        infoMsg.textContent = `✅ 生成成功，共识别 ${boxes1.length} 个出库箱号，${boxes2.length} 个库存箱号`;

        // 渲染表格（展示出库单，高亮箱号列）
        renderTable(file1Data, headers1, boxIndex1);

        // 汇总信息
        document.getElementById('extractedDate').textContent = dateValue;
        document.getElementById('totalBoxes').textContent = boxes1.length;
        document.getElementById('totalBoxesStock').textContent = boxes2.length;
        const match = boxes1.length === boxes2.length ? '✅ 相符' : '❌ 不符，请检查';
        document.getElementById('checkResult').textContent = match;
        document.getElementById('checkResult').style.color = boxes1.length === boxes2.length ? '#16a34a' : '#b91c1c';
        document.getElementById('colPosition').textContent = `第 ${boxIndex1+1} 列 (索引 ${boxIndex1})`;

        // 启用下载按钮
        downloadBtn.style.display = 'inline-block';
    });

    // 渲染表格
    function renderTable(data, headers, boxIdx) {
        tableHead.innerHTML = '';
        tableBody.innerHTML = '';

        // 表头
        const thead = document.createElement('tr');
        headers.forEach((h, idx) => {
            const th = document.createElement('th');
            th.textContent = h;
            if (idx === boxIdx) {
                th.style.background = '#e6f4ff';
                th.style.borderLeft = '3px solid #3b82f6';
                th.style.borderRight = '3px solid #3b82f6';
                th.innerHTML += ' 🎯';
            }
            thead.appendChild(th);
        });
        tableHead.appendChild(thead);

        // 数据行（最多显示前100行）
        const rows = data.slice(0, 100);
        rows.forEach(row => {
            const tr = document.createElement('tr');
            headers.forEach((h, idx) => {
                const td = document.createElement('td');
                let val = row[h] !== undefined && row[h] !== null ? String(row[h]) : '';
                td.textContent = val;
                if (idx === boxIdx) {
                    td.className = 'col-box-highlight';
                }
                tr.appendChild(td);
            });
            tableBody.appendChild(tr);
        });
        if (data.length > 100) {
            const tr = document.createElement('tr');
            const td = document.createElement('td');
            td.colSpan = headers.length;
            td.textContent = `... 仅显示前100行，共 ${data.length} 行`;
            td.style.textAlign = 'center';
            td.style.color = '#6b7b93';
            tr.appendChild(td);
            tableBody.appendChild(tr);
        }
    }

    // ============================================================
    //  导出 Excel（核心新增功能）
    // ============================================================
    downloadBtn.addEventListener('click', function() {
        if (!currentExportData || currentExportData.length === 0) {
            alert('没有可导出的数据，请先生成拣货单。');
            return;
        }

        try {
            // 创建工作簿
            const wb = XLSX.utils.book_new();
            // 将数据转换为数组（包含表头）
            const wsData = [currentExportHeaders];
            currentExportData.forEach(row => {
                const rowArray = currentExportHeaders.map(key => row[key] !== undefined ? row[key] : '');
                wsData.push(rowArray);
            });

            const ws = XLSX.utils.aoa_to_sheet(wsData);
            XLSX.utils.book_append_sheet(wb, ws, 'Picking List');

            // 文件名：使用日期 + 固定前缀
            let fileName = 'Picking_List';
            if (currentDateValue && currentDateValue !== '—') {
                fileName += `_${currentDateValue}`;
            } else {
                // 若没有日期，使用当前时间
                const now = new Date();
                const dateStr = now.toISOString().slice(0,10);
                fileName += `_${dateStr}`;
            }
            fileName += '.xlsx';

            // 导出
            XLSX.writeFile(wb, fileName);
            infoMsg.textContent = `✅ 文件已导出：${fileName}`;
        } catch (err) {
            alert('导出失败，请检查数据或重试。');
            console.error(err);
        }
    });

    // 初始隐藏下载按钮（未生成时不可用）
    downloadBtn.style.display = 'none';

    console.log('智能拣货单系统已加载，支持导出Excel。');
