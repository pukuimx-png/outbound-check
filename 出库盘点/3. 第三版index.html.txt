        (function() {
            'use strict';

            // ---- DOM ----
            const scanInput = document.getElementById('scanInput');
            const scanBtn = document.getElementById('scanBtn');
            const listBody = document.getElementById('listBody');
            const modeDisplay = document.getElementById('modeDisplay');
            const myKeysDisplay = document.getElementById('myKeysDisplay');
            const myTotalBox = document.getElementById('myTotalBox');
            const mySubProgress = document.getElementById('mySubProgress');
            const operatorDisplay = document.getElementById('operatorDisplay');
            const vehicleDisplay = document.getElementById('vehicleDisplay');
            const endMainBtn = document.getElementById('endMainBtn');
            const resetBatchBtn = document.getElementById('resetBatchBtn');
            const resetAllBtn = document.getElementById('resetAllBtn');
            const uploadZone = document.getElementById('uploadZone');
            const fileInput = document.getElementById('fileInput');
            const fileInfo = document.getElementById('fileInfo');
            const exportTxtBtn = document.getElementById('exportTxtBtn');
            const exportExcelBtn = document.getElementById('exportExcelBtn');
            const connectionStatus = document.getElementById('connectionStatus');
            const operatorModal = document.getElementById('operatorModal');
            const operatorInput = document.getElementById('operatorInput');
            const vehicleInput = document.getElementById('vehicleInput');
            const confirmOperatorBtn = document.getElementById('confirmOperatorBtn');
            const toastEl = document.getElementById('toast');

            // ---- 状态 ----
            let ws = null;
            let operator = '';
            let vehicle = '';
            let localMode = 'main';
            let localAddedKeys = [];
            let state = {
                groups: [],
                selectedMainKeys: [],
                scanStatus: {}
            };
            let reconnectTimer = null;
            let toastTimer = null;
            const WS_URL = `ws://${window.location.hostname}:3000`;

            // ---- Toast ----
            function showToast(msg, type = 'info', duration = 3000) {
                if (toastTimer) clearTimeout(toastTimer);
                toastEl.textContent = msg;
                toastEl.className = 'toast';
                if (type === 'success') toastEl.classList.add('success');
                else if (type === 'error') toastEl.classList.add('error');
                else if (type === 'warning') toastEl.classList.add('warning');
                toastEl.classList.add('show');
                toastTimer = setTimeout(() => {
                    toastEl.classList.remove('show');
                }, duration);
                console.log(`[Toast] ${type}: ${msg}`);
            }

            // ---- 统计 ----
            function getMyStats() {
                let total = 0,
                    scanned = 0;
                for (const key of localAddedKeys) {
                    const g = state.groups.find(gr => gr.key === key);
                    if (g) {
                        total += g.items.length;
                        for (const item of g.items) {
                            if (state.scanStatus[item.code]) scanned++;
                        }
                    }
                }
                return { total, scanned };
            }

            // ---- 更新状态栏 ----
            function updateStatusBar() {
                modeDisplay.textContent = localMode === 'main' ? '主码' : '子码';
                myKeysDisplay.textContent = localAddedKeys.length ? localAddedKeys.join(', ') : '无';
                const stats = getMyStats();
                myTotalBox.textContent = stats.total;
                mySubProgress.textContent = `${stats.scanned}/${stats.total}`;
                endMainBtn.disabled = (localMode === 'sub' || localAddedKeys.length === 0);

                // 如果子码模式且全部扫完，自动切回主码
                if (localMode === 'sub' && stats.total > 0 && stats.scanned === stats.total) {
                    showToast('🎉 您的所有子码扫描完成！自动切回主码模式', 'success', 3000);
                    localMode = 'main';
                    updateStatusBar();
                    renderList();
                }
            }

            // ---- WebSocket ----
            function connectWebSocket() {
                if (ws && ws.readyState === WebSocket.OPEN) return;
                try {
                    ws = new WebSocket(WS_URL);
                    ws.onopen = function() {
                        connectionStatus.textContent = '● 已连接';
                        connectionStatus.style.color = '#16a34a';
                        showToast('✅ 已连接到服务器', 'success', 2000);
                        console.log('✅ WebSocket 连接成功');
                    };
                    ws.onmessage = function(event) {
                        try {
                            const data = JSON.parse(event.data);
                            console.log('📨 收到服务器消息:', data);
                            if (data.type === 'state') {
                                state = data.state;
                                // 同步本地列表：如果本地主码不在全局列表中，则移除
                                localAddedKeys = localAddedKeys.filter(k => state.selectedMainKeys.includes(k));
                                renderList();
                                updateStatusBar();
                            } else if (data.type === 'success') {
                                // 如果是主码添加成功，将主码加入本地列表
                                const msg = data.message;
                                if (msg.includes('已添加主码:')) {
                                    const match = msg.match(/已添加主码:\s*([^\s]+)/);
                                    if (match) {
                                        const key = match[1];
                                        if (!localAddedKeys.includes(key)) {
                                            localAddedKeys.push(key);
                                            // 更新状态栏和清单
                                            renderList();
                                            updateStatusBar();
                                        }
                                    }
                                }
                                showToast(data.message, 'success', 2000);
                                scanInput.value = '';
                                scanInput.focus();
                            } else if (data.type === 'error') {
                                showToast(data.message, 'error', 3000);
                                scanInput.value = '';
                                scanInput.focus();
                            }
                        } catch (err) {
                            console.error('❌ 解析消息失败:', err);
                        }
                    };
                    ws.onclose = function() {
                        connectionStatus.textContent = '● 未连接';
                        connectionStatus.style.color = '#dc2626';
                        showToast('⚠️ 连接断开，尝试重连...', 'error');
                        if (reconnectTimer) clearTimeout(reconnectTimer);
                        reconnectTimer = setTimeout(connectWebSocket, 3000);
                    };
                    ws.onerror = function(err) {
                        console.error('❌ WebSocket错误:', err);
                        ws.close();
                    };
                } catch (err) {
                    console.error('❌ 连接失败:', err);
                    setTimeout(connectWebSocket, 3000);
                }
            }

            // ---- 发送扫描 ----
            function sendScan(code) {
                if (!code || code.trim() === '') {
                    showToast('请输入条码', 'warning');
                    return;
                }
                if (!ws || ws.readyState !== WebSocket.OPEN) {
                    showToast('❌ 服务器未连接', 'error');
                    return;
                }
                const trimmed = code.trim();
                if (localMode === 'main') {
                    ws.send(JSON.stringify({ type: 'scan', code: trimmed, mode: 'main' }));
                } else {
                    // 子码模式：先校验是否属于本地主码
                    let belongs = false;
                    for (const key of localAddedKeys) {
                        const g = state.groups.find(gr => gr.key === key);
                        if (g) {
                            if (g.items.find(i => i.code === trimmed)) { belongs = true; break; }
                            const parts = trimmed.split('/');
                            if (parts.length === 2 && parts[0] === key) {
                                const idx = parseInt(parts[1]);
                                if (!isNaN(idx) && idx >= 1 && idx <= g.items.length) { belongs = true; break; }
                            }
                        }
                    }
                    if (!belongs) {
                        showToast(`❌ 子码 ${trimmed} 不属于您添加的主码`, 'error', 3000);
                        scanInput.value = '';
                        scanInput.focus();
                        return;
                    }
                    ws.send(JSON.stringify({ type: 'scan', code: trimmed, mode: 'sub' }));
                }
            }

            // ---- 切换子码模式 ----
            function switchToSubMode() {
                if (localAddedKeys.length === 0) {
                    showToast('您还没有添加任何主码', 'warning');
                    return;
                }
                if (localMode === 'sub') {
                    showToast('已经是子码模式', 'warning');
                    return;
                }
                localMode = 'sub';
                updateStatusBar();
                renderList();
                showToast('🔁 已切换至子码模式，仅扫描您添加的主码的子码', 'success', 2000);
                scanInput.focus();
            }

            // ---- 重置我的主码（释放并清空本地） ----
            function resetMyKeys() {
                if (localAddedKeys.length === 0) {
                    showToast('您没有主码需要重置', 'warning');
                    return;
                }
                if (!ws || ws.readyState !== WebSocket.OPEN) {
                    showToast('❌ 服务器未连接', 'error');
                    return;
                }
                // 发送释放请求
                const keysToRelease = [...localAddedKeys];
                ws.send(JSON.stringify({ type: 'release_keys', keys: keysToRelease }));
                // 本地立即清空并切回主码（服务器会广播，我们收到更新后会同步）
                localAddedKeys = [];
                localMode = 'main';
                updateStatusBar();
                renderList();
                showToast(`🔄 已释放主码: ${keysToRelease.join(', ')}，其他设备可重新添加`, 'info', 3000);
                scanInput.focus();
            }

            // ---- 全部重置 ----
            function resetAll() {
                if (!ws || ws.readyState !== WebSocket.OPEN) {
                    showToast('❌ 服务器未连接', 'error');
                    return;
                }
                if (confirm('确定重置所有扫描记录吗？（将清除所有主码和扫描状态）')) {
                    ws.send(JSON.stringify({ type: 'reset_all' }));
                    localAddedKeys = [];
                    localMode = 'main';
                    updateStatusBar();
                }
            }

            // ---- 渲染清单 ----
            function renderList() {
                const groups = state.groups;
                if (groups.length === 0) {
                    listBody.innerHTML = `<div style="padding:30px;text-align:center;color:#94a3b8;">暂无数据，请上传文件</div>`;
                    return;
                }

                let html = '';
                let globalIdx = 0;
                const mySet = new Set(localAddedKeys);

                for (const g of groups) {
                    const isMy = mySet.has(g.key);
                    const allScanned = g.items.every(item => state.scanStatus[item.code] || false);
                    const typeLabel = g.type === 'fba' ? 'FBA' : '自定义';
                    const typeClass = g.type === 'fba' ? 'fba' : 'custom';

                    globalIdx++;
                    html += `
                        <div class="list-item group-header ${isMy ? 'active' : ''}">
                            <span>${globalIdx}</span>
                            <span class="code">${g.key} <span class="badge">${g.items.length}箱</span> ${isMy ? '⭐ (我的)' : ''}</span>
                            <span class="badge ${typeClass}">${typeLabel}</span>
                            <span class="status">${allScanned ? '✅' : '⬜'}</span>
                        </div>
                    `;

                    for (let i = 0; i < g.items.length; i++) {
                        const item = g.items[i];
                        globalIdx++;
                        const scanned = state.scanStatus[item.code] || false;
                        let displayCode = item.code;
                        if (g.type === 'fba') {
                            displayCode = `${g.key}/${i+1}`;
                        }
                        const isMyItem = isMy && localMode === 'sub';
                        html += `
                            <div class="list-item ${scanned ? 'scanned' : ''} ${isMyItem ? 'active' : ''}">
                                <span>${globalIdx}</span>
                                <span>${displayCode}</span>
                                <span class="badge" style="opacity:0.5;">子项</span>
                                <span class="status">${scanned ? '✅' : '⬜'}</span>
                            </div>
                        `;
                    }
                }
                listBody.innerHTML = html;
            }

            // ---- 上传文件 ----
            function handleFile(file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    try {
                        const data = new Uint8Array(e.target.result);
                        const workbook = XLSX.read(data, { type: 'array' });
                        const sheet = workbook.Sheets[workbook.SheetNames[0]];
                        const rows = XLSX.utils.sheet_to_json(sheet, { defval: '' });

                        const parsed = parseExcelData(rows);
                        if (parsed && parsed.length > 0) {
                            if (ws && ws.readyState === WebSocket.OPEN) {
                                ws.send(JSON.stringify({ type: 'init', groups: parsed }));
                                fileInfo.textContent = `✅ 已上传: ${file.name} (${parsed.length} 个分组)`;
                                showToast('📤 数据已发送至服务器', 'success');
                                state.groups = parsed;
                            } else {
                                showToast('❌ 服务器未连接', 'error');
                            }
                        }
                    } catch (err) {
                        showToast('❌ 解析失败: ' + err.message, 'error');
                        console.error('解析错误:', err);
                    }
                };
                reader.readAsArrayBuffer(file);
            }

            function parseExcelData(rows) {
                let kCol = null,
                    asCol = null,
                    statusCol = null;
                const headers = Object.keys(rows[0] || {});
                for (const h of headers) {
                    const clean = h.trim();
                    if (clean.includes('FBA货件ID') || clean.includes('FBAShipmentID') || clean.includes('货件ID')) kCol = h;
                    if (clean.includes('Custom box barcode') || clean.includes('自定义箱条码') || clean.includes('箱条码')) asCol =
                    h;
                    if (clean.includes('Status') || clean.includes('状态')) statusCol = h;
                }
                if (!kCol && !asCol) {
                    showToast('未找到K列或AS列', 'error');
                    return null;
                }

                const raw = [];
                for (const row of rows) {
                    let kVal = kCol ? String(row[kCol] || '').trim() : '';
                    let asVal = asCol ? String(row[asCol] || '').trim() : '';
                    let statusVal = statusCol ? String(row[statusCol] || '').trim() : '';
                    let code = kVal || asVal;
                    if (!code) continue;
                    let type = 'fba',
                        prefix = '';
                    if (kVal) { type = 'fba';
                        prefix = kVal; } else if (asVal) { type = 'custom';
                        prefix = extractPrefix(asVal); } else continue;
                    raw.push({ k: kVal, as: asVal, code: asVal || kVal, prefix, type, status: statusVal });
                }

                const map = new Map();
                for (const item of raw) {
                    const key = item.prefix;
                    const gkey = key + '|' + item.type;
                    if (!map.has(gkey)) {
                        map.set(gkey, { key, type: item.type, items: [], hasCanceled: false });
                    }
                    const g = map.get(gkey);
                    g.items.push({ code: item.code, raw: item });
                    if (item.status === '已取消') g.hasCanceled = true;
                }
                const groups = [];
                for (const [_, g] of map) {
                    g.items.sort((a, b) => {
                        const na = parseInt(a.code.replace(/^.*?(\d+)$/, '$1'));
                        const nb = parseInt(b.code.replace(/^.*?(\d+)$/, '$1'));
                        if (!isNaN(na) && !isNaN(nb)) return na - nb;
                        return a.code.localeCompare(b.code);
                    });
                    groups.push(g);
                }
                groups.sort((a, b) => {
                    if (a.type !== b.type) return a.type === 'fba' ? -1 : 1;
                    return a.key.localeCompare(b.key);
                });
                return groups;
            }

            function extractPrefix(val) {
                if (!val) return val;
                const s = String(val).trim();
                if (s.includes('/')) return s.split('/')[0].trim();
                const m = s.match(/^(.*?[A-Za-z])U\d+$/);
                if (m) return m[1] + 'U';
                const m2 = s.match(/^(.*?[A-Za-z])\d+$/);
                if (m2 && m2[1].length >= 2) return m2[1];
                return s;
            }

            // ---- 导出 ----
            function exportTxt() {
                const lines = [];
                lines.push('出库复核清单');
                lines.push(`生成: ${new Date().toLocaleString()}`);
                lines.push('─'.repeat(50));
                for (const g of state.groups) {
                    const typeLabel = g.type === 'fba' ? 'FBA货件' : '自定义箱条码';
                    lines.push(`【${typeLabel}: ${g.key}】 ${g.items.length} 箱`);
                    for (let i = 0; i < g.items.length; i++) {
                        const item = g.items[i];
                        let displayCode = item.code;
                        if (g.type === 'fba') displayCode = `${g.key}/${i+1}`;
                        const scanned = state.scanStatus[item.code] || false;
                        lines.push(`  ${String(i+1).padStart(3,' ')}. ${scanned ? '✅' : '⬜'} ${displayCode}`);
                    }
                    lines.push('');
                }
                const blob = new Blob([lines.join('\n')], { type: 'text/plain;charset=utf-8' });
                const a = document.createElement('a');
                a.href = URL.createObjectURL(blob);
                a.download = `复核清单_${new Date().toISOString().slice(0,10)}.txt`;
                a.click();
                URL.revokeObjectURL(a.href);
                showToast('📄 TXT已导出', 'success');
            }

            function exportExcel() {
                const rows = [
                    ['#', '分组类型', '分组Key', '子项序号', '子项条码', '已扫描']
                ];
                let idx = 0;
                for (const g of state.groups) {
                    const typeLabel = g.type === 'fba' ? 'FBA货件' : '自定义箱条码';
                    idx++;
                    rows.push([idx, typeLabel, g.key, '', `共${g.items.length}箱`, '']);
                    for (let i = 0; i < g.items.length; i++) {
                        const item = g.items[i];
                        let displayCode = item.code;
                        if (g.type === 'fba') displayCode = `${g.key}/${i+1}`;
                        const scanned = state.scanStatus[item.code] || false;
                        rows.push(['', '', '', i + 1, displayCode, scanned ? '是' : '否']);
                    }
                }
                const ws = XLSX.utils.aoa_to_sheet(rows);
                const wb = XLSX.utils.book_new();
                XLSX.utils.book_append_sheet(wb, ws, '复核清单');
                const out = XLSX.write(wb, { bookType: 'xlsx', type: 'array' });
                const blob = new Blob([out], { type: 'application/octet-stream' });
                const a = document.createElement('a');
                a.href = URL.createObjectURL(blob);
                a.download = `复核清单_${new Date().toISOString().slice(0,10)}.xlsx`;
                a.click();
                URL.revokeObjectURL(a.href);
                showToast('📊 Excel已导出', 'success');
            }

            }
