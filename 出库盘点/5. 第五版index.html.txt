    <script>
        (function() {
            'use strict';

            // ---- DOM ----
            const scanInput = document.getElementById('scanInput');
            const scanBtn = document.getElementById('scanBtn');
            const listBody = document.getElementById('listBody');
            const modeDisplay = document.getElementById('modeDisplay');
            const taskKeysDisplay = document.getElementById('taskKeysDisplay');
            const pendingDisplay = document.getElementById('pendingDisplay');
            const mySubProgress = document.getElementById('mySubProgress');
            const globalMainCount = document.getElementById('globalMainCount');
            const globalTotalBox = document.getElementById('globalTotalBox');
            const globalScanned = document.getElementById('globalScanned');
            const globalUnscanned = document.getElementById('globalUnscanned');
            const operatorDisplay = document.getElementById('operatorDisplay');
            const vehicleDisplay = document.getElementById('vehicleDisplay');
            const endMainBtn = document.getElementById('endMainBtn');
            const forceEndBtn = document.getElementById('forceEndBtn');
            const resetBatchBtn = document.getElementById('resetBatchBtn');
            const resetAllBtn = document.getElementById('resetAllBtn');
            const uploadZone = document.getElementById('uploadZone');
            const fileInput = document.getElementById('fileInput');
            const fileInfo = document.getElementById('fileInfo');
            const exportTxtBtn = document.getElementById('exportTxtBtn');
            const exportExcelBtn = document.getElementById('exportExcelBtn');
            const connectionStatus = document.getElementById('connectionStatus');
            const operatorModal = document.getElementById('operatorModal');
            const operatorInput = document.getElementById('operatorInput');
            const vehicleInput = document.getElementById('vehicleInput');
            const confirmOperatorBtn = document.getElementById('confirmOperatorBtn');
            const toastEl = document.getElementById('toast');
            const globalStatsToggle = document.getElementById('globalStatsToggle');
            const globalStatsContent = document.getElementById('globalStatsContent');
            const globalStatsIcon = document.getElementById('globalStatsIcon');
            const toggleDoneBtn = document.getElementById('toggleDoneBtn');
            const resetCollapseToggle = document.getElementById('resetCollapseToggle');
            const resetCollapseBody = document.getElementById('resetCollapseBody');
            const resetCollapseArrow = document.getElementById('resetCollapseArrow');

            // ---- 状态 ----
            let ws = null;
            let operator = '';
            let vehicle = '';
            let localMode = 'main';
            let taskKeys = [];
            let pendingKeys = [];
            let state = {
                groups: [],
                selectedMainKeys: [],
                scanStatus: {}
            };
            let reconnectTimer = null;
            let toastTimer = null;
            const WS_URL = `ws://${window.location.hostname}:3000`;

            // ---- UI 状态 ----
            let globalStatsCollapsed = false;
            let showOnlyDone = false;
            let resetCollapsed = true;

            // ---- 持久化 ----
            function saveLocalState() {
                try {
                    localStorage.setItem('pda_localMode', localMode);
                    localStorage.setItem('pda_taskKeys', JSON.stringify(taskKeys));
                    localStorage.setItem('pda_pendingKeys', JSON.stringify(pendingKeys));
                    localStorage.setItem('pda_showOnlyDone', JSON.stringify(showOnlyDone));
                    localStorage.setItem('pda_statsCollapsed', JSON.stringify(globalStatsCollapsed));
                    localStorage.setItem('pda_resetCollapsed', JSON.stringify(resetCollapsed));
                } catch (e) { console.warn('保存本地状态失败:', e); }
            }

            function loadLocalState() {
                try {
                    const mode = localStorage.getItem('pda_localMode');
                    const task = localStorage.getItem('pda_taskKeys');
                    const pending = localStorage.getItem('pda_pendingKeys');
                    const done = localStorage.getItem('pda_showOnlyDone');
                    const collapsed = localStorage.getItem('pda_statsCollapsed');
                    const resetCol = localStorage.getItem('pda_resetCollapsed');
                    if (mode) localMode = mode;
                    if (task) { const p = JSON.parse(task); if (Array.isArray(p)) taskKeys = p; }
                    if (pending) { const p = JSON.parse(pending); if (Array.isArray(p)) pendingKeys = p; }
                    if (done) showOnlyDone = JSON.parse(done);
                    if (collapsed) globalStatsCollapsed = JSON.parse(collapsed);
                    if (resetCol) resetCollapsed = JSON.parse(resetCol);
                } catch (e) { console.warn('加载本地状态失败:', e); }
            }

            // ---- Toast ----
            function showToast(msg, type = 'info', duration = 3000) {
                if (toastTimer) clearTimeout(toastTimer);
                toastEl.textContent = msg;
                toastEl.className = 'toast';
                if (type === 'success') toastEl.classList.add('success');
                else if (type === 'error') toastEl.classList.add('error');
                else if (type === 'warning') toastEl.classList.add('warning');
                toastEl.classList.add('show');
                toastTimer = setTimeout(() => {
                    toastEl.classList.remove('show');
                }, duration);
                console.log(`[Toast] ${type}: ${msg}`);
            }

            // ---- 统计 ----
            function getTaskStats() {
                let total = 0,
                    scanned = 0;
                for (const key of taskKeys) {
                    const g = state.groups.find(gr => gr.key === key);
                    if (g) {
                        total += g.items.length;
                        for (const item of g.items) {
                            if (state.scanStatus[item.code]) scanned++;
                        }
                    }
                }
                return { total, scanned };
            }

            function getGlobalStats() {
                const totalMain = state.selectedMainKeys.length;
                let totalBoxes = 0,
                    scannedBoxes = 0;
                for (const key of state.selectedMainKeys) {
                    const g = state.groups.find(gr => gr.key === key);
                    if (g) {
                        totalBoxes += g.items.length;
                        for (const item of g.items) {
                            if (state.scanStatus[item.code]) scannedBoxes++;
                        }
                    }
                }
                return { totalMain, totalBoxes, scannedBoxes, unscannedBoxes: totalBoxes - scannedBoxes };
            }

            // ---- 检查主码是否已完成 ----
            function isMainCompleted(key) {
                const g = state.groups.find(gr => gr.key === key);
                if (!g) return false;
                return g.items.every(item => state.scanStatus[item.code] || false);
            }

            // ---- 更新状态栏 ----
            function updateStatusBar() {
                modeDisplay.textContent = localMode === 'main' ? '主码' : '子码';
                taskKeysDisplay.textContent = taskKeys.length ? taskKeys.join(', ') : '无';
                pendingDisplay.textContent = pendingKeys.length ? pendingKeys.join(', ') : '无';
                const stats = getTaskStats();
                mySubProgress.textContent = `${stats.scanned}/${stats.total}`;
                endMainBtn.disabled = (localMode === 'sub' || pendingKeys.length === 0);
                forceEndBtn.disabled = (localMode === 'main' || taskKeys.length === 0);

                const global = getGlobalStats();
                globalMainCount.textContent = global.totalMain;
                globalTotalBox.textContent = global.totalBoxes;
                globalScanned.textContent = global.scannedBoxes;
                globalUnscanned.textContent = global.unscannedBoxes;

                if (globalStatsCollapsed) {
                    globalStatsContent.classList.add('collapsed');
                    globalStatsIcon.classList.add('collapsed');
                    globalStatsIcon.textContent = '▶';
                } else {
                    globalStatsContent.classList.remove('collapsed');
                    globalStatsIcon.classList.remove('collapsed');
                    globalStatsIcon.textContent = '▼';
                }

                if (resetCollapsed) {
                    resetCollapseBody.classList.remove('open');
                    resetCollapseArrow.classList.remove('open');
                    resetCollapseArrow.textContent = '▶';
                } else {
                    resetCollapseBody.classList.add('open');
                    resetCollapseArrow.classList.add('open');
                    resetCollapseArrow.textContent = '▼';
                }

                saveLocalState();

                // 子码自动完成
                if (localMode === 'sub' && taskKeys.length > 0 && stats.total > 0 && stats.scanned === stats.total) {
                    showToast('🎉 当前任务所有子码扫描完成！自动切回主码模式', 'success', 3000);
                    localMode = 'main';
                    taskKeys = [];
                    updateStatusBar();
                    renderList();
                }
            }

            // ---- WebSocket ----
            function connectWebSocket() {
                if (ws && ws.readyState === WebSocket.OPEN) return;
                try {
                    ws = new WebSocket(WS_URL);
                    ws.onopen = function() {
                        connectionStatus.textContent = '● 已连接';
                        connectionStatus.style.color = '#16a34a';
                        showToast('✅ 已连接到服务器', 'success', 2000);
                        console.log('✅ WebSocket 连接成功');
                    };
                    ws.onmessage = function(event) {
                        try {
                            const data = JSON.parse(event.data);
                            console.log('📨 收到服务器消息:', data);
                            if (data.type === 'state') {
                                state = data.state;
                                // 过滤掉已被释放的主码
                                taskKeys = taskKeys.filter(k => state.selectedMainKeys.includes(k));
                                pendingKeys = pendingKeys.filter(k => state.selectedMainKeys.includes(k));
                                if (localMode === 'sub' && taskKeys.length === 0) {
                                    localMode = 'main';
                                    showToast('🔄 您的任务已失效，已切回主码模式', 'warning', 2000);
                                }
                                renderList();
                                updateStatusBar();
                            } else if (data.type === 'success') {
                                const msg = data.message;
                                console.log('🔍 成功消息内容:', msg);
                                // 更可靠的主码提取
                                const match = msg.match(/已添加主码:\s*([^\s]+)/);
                                if (match) {
                                    const key = match[1];
                                    console.log('✅ 提取到主码:', key);
                                    // 检查是否已完成
                                    if (isMainCompleted(key)) {
                                        showToast(`⏳ 主码 ${key} 已完成，无需再次添加`, 'warning', 2000);
                                        scanInput.value = '';
                                        scanInput.focus();
                                        return;
                                    }
                                    if (!pendingKeys.includes(key)) {
                                        pendingKeys.push(key);
                                        console.log('📌 pendingKeys 已更新:', pendingKeys);
                                        renderList();
                                        updateStatusBar();
                                    } else {
                                        console.log('⏳ 主码已在待添加列表中');
                                    }
                                } else {
                                    console.warn('⚠️ 未匹配到主码，消息原文:', msg);
                                }
                                showToast(data.message, 'success', 2000);
                                scanInput.value = '';
                                scanInput.focus();
                            } else if (data.type === 'error') {
                                showToast(data.message, 'error', 3000);
                                scanInput.value = '';
                                scanInput.focus();
                            }
                        } catch (err) {
                            console.error('❌ 解析消息失败:', err);
                        }
                    };
                    ws.onclose = function() {
                        connectionStatus.textContent = '● 未连接';
                        connectionStatus.style.color = '#dc2626';
                        showToast('⚠️ 连接断开，尝试重连...', 'error');
                        if (reconnectTimer) clearTimeout(reconnectTimer);
                        reconnectTimer = setTimeout(connectWebSocket, 3000);
                    };
                    ws.onerror = function(err) {
                        console.error('❌ WebSocket错误:', err);
                        ws.close();
                    };
                } catch (err) {
                    console.error('❌ 连接失败:', err);
                    setTimeout(connectWebSocket, 3000);
                }
            }

            // ---- 发送扫描 ----
            function sendScan(code) {
                if (!code || code.trim() === '') {
                    showToast('请输入条码', 'warning');
                    return;
                }
                if (!ws || ws.readyState !== WebSocket.OPEN) {
                    showToast('❌ 服务器未连接', 'error');
                    return;
                }
                const trimmed = code.trim();
                if (localMode === 'main') {
                    console.log('📤 发送主码扫描:', trimmed);
                    ws.send(JSON.stringify({ type: 'scan', code: trimmed, mode: 'main' }));
                } else {
                    // 子码模式：必须属于 taskKeys
                    let belongs = false;
                    for (const key of taskKeys) {
                        const g = state.groups.find(gr => gr.key === key);
                        if (g) {
                            if (g.items.find(i => i.code === trimmed)) { belongs = true; break; }
                            const parts = trimmed.split('/');
                            if (parts.length === 2 && parts[0] === key) {
                                const idx = parseInt(parts[1]);
                                if (!isNaN(idx) && idx >= 1 && idx <= g.items.length) { belongs = true; break; }
                            }
                        }
                    }
                    if (!belongs) {
                        showToast(`❌ 子码 ${trimmed} 不属于当前任务主码`, 'error', 3000);
                        scanInput.value = '';
                        scanInput.focus();
                        return;
                    }
                    console.log('📤 发送子码扫描:', trimmed);
                    ws.send(JSON.stringify({ type: 'scan', code: trimmed, mode: 'sub' }));
                }
            }

            // ---- 结束主码：用 pendingKeys 覆盖 taskKeys ----
            function switchToSubMode() {
                if (pendingKeys.length === 0) {
                    showToast('您还没有添加任何主码', 'warning');
                    return;
                }
                if (localMode === 'sub') {
                    showToast('已经是子码模式', 'warning');
                    return;
                }
                // 覆盖任务
                taskKeys = [...pendingKeys];
                pendingKeys = [];
                localMode = 'sub';
                updateStatusBar();
                renderList();
                showToast(`🔁 已切换至子码模式，当前任务: ${taskKeys.join(', ')}`, 'success', 2000);
                scanInput.focus();
            }

            // ---- 强制结束 ----
            function forceEnd() {
                if (localMode === 'main') {
                    showToast('当前已是主码模式', 'warning');
                    return;
                }
                if (taskKeys.length === 0) {
                    showToast('没有任务需要结束', 'warning');
                    return;
                }
                const stats = getTaskStats();
                if (stats.scanned === stats.total) {
                    showToast('任务已全部完成，无需强制结束', 'info', 2000);
                    return;
                }
                if (!confirm(`确定强制结束吗？\n剩余 ${stats.total - stats.scanned} 个子码未扫，任务将保留以便补扫。`)) {
                    return;
                }
                localMode = 'main';
                updateStatusBar();
                renderList();
                showToast(`⏹ 已强制结束，剩余 ${stats.total - stats.scanned} 个子码未扫，可后续补扫`, 'warning', 4000);
                scanInput.focus();
            }

            // ---- 重置我的主码 ----
            function resetMyKeys() {
                const allKeys = [...new Set([...taskKeys, ...pendingKeys])];
                if (allKeys.length === 0) {
                    showToast('您没有主码需要重置', 'warning');
                    return;
                }
                if (!ws || ws.readyState !== WebSocket.OPEN) {
                    showToast('❌ 服务器未连接', 'error');
                    return;
                }
                if (!confirm(`确定释放所有主码吗？\n将释放: ${allKeys.join(', ')}`)) return;
                ws.send(JSON.stringify({ type: 'release_keys', keys: allKeys }));
                taskKeys = [];
                pendingKeys = [];
                localMode = 'main';
                updateStatusBar();
                renderList();
                showToast(`🔄 已释放主码: ${allKeys.join(', ')}`, 'info', 3000);
                scanInput.focus();
            }

            // ---- 全部重置 ----
            function resetAll() {
                if (!ws || ws.readyState !== WebSocket.OPEN) {
                    showToast('❌ 服务器未连接', 'error');
                    return;
                }
                if (confirm('确定重置所有扫描记录吗？（将清除所有主码和扫描状态）')) {
                    ws.send(JSON.stringify({ type: 'reset_all' }));
                    taskKeys = [];
                    pendingKeys = [];
                    localMode = 'main';
                    updateStatusBar();
                }
            }

            // ---- 渲染清单 ----
            function renderList() {
                const groups = state.groups;
                if (groups.length === 0) {
                    listBody.innerHTML = `<div style="padding:30px;text-align:center;color:#94a3b8;">暂无数据，请上传文件</div>`;
                    return;
                }

                let html = '';
                let globalIdx = 0;
                const taskSet = new Set(taskKeys);
                const pendingSet = new Set(pendingKeys);

                for (const g of groups) {
                    const isTask = taskSet.has(g.key);
                    const isPending = pendingSet.has(g.key);
                    const allScanned = g.items.every(item => state.scanStatus[item.code] || false);
                    const typeLabel = g.type === 'fba' ? 'FBA' : '自定义';
                    const typeClass = g.type === 'fba' ? 'fba' : 'custom';

                    if (showOnlyDone && !allScanned) continue;

                    globalIdx++;
                    let statusText = '';
                    if (isTask) statusText = '📋 任务中';
                    else if (isPending) statusText = '⏳ 待添加';
                    else statusText = '';
                    const isActive = isTask && localMode === 'sub';
                    html += `
                        <div class="list-item group-header ${isActive ? 'active' : ''}">
                            <span>${globalIdx}</span>
                            <span class="code">${g.key} <span class="badge">${g.items.length}箱</span> ${statusText} ${allScanned ? '✅ 完成' : ''}</span>
                            <span class="badge ${typeClass}">${typeLabel}</span>
                            <span class="status">${allScanned ? '✅' : '⬜'}</span>
                        </div>
                    `;

                    for (let i = 0; i < g.items.length; i++) {
                        const item = g.items[i];
                        globalIdx++;
                        const scanned = state.scanStatus[item.code] || false;
                        let displayCode = item.code;
                        if (g.type === 'fba') {
                            displayCode = `${g.key}/${i+1}`;
                        }
                        const isActiveItem = isTask && localMode === 'sub';
                        html += `
                            <div class="list-item ${scanned ? 'scanned' : ''} ${isActiveItem ? 'active' : ''}">
                                <span>${globalIdx}</span>
                                <span>${displayCode}</span>
                                <span class="badge" style="opacity:0.5;">子项</span>
                                <span class="status">${scanned ? '✅' : '⬜'}</span>
                            </div>
                        `;
                    }
                }
                if (html === '') {
                    html = `<div style="padding:20px;text-align:center;color:#94a3b8;">${showOnlyDone ? '暂无已完成的分组' : '没有匹配的分组'}</div>`;
                }
                listBody.innerHTML = html;
            }

            // ---- 切换显示模式 ----
            function toggleShowOnlyDone() {
                showOnlyDone = !showOnlyDone;
                if (showOnlyDone) {
                    toggleDoneBtn.classList.add('active');

            }

            // ---- 上传文件 ----
            function handleFile(file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    try {
                        const data = new Uint8Array(e.target.result);
                        const workbook = XLSX.read(data, { type: 'array' });
                        const sheet = workbook.Sheets[workbook.SheetNames[0]];
                        const rows = XLSX.utils.sheet_to_json(sheet, { defval: '' });

                        const parsed = parseExcelData(rows);
                        if (parsed && parsed.length > 0) {
                            if (ws && ws.readyState === WebSocket.OPEN) {
                                ws.send(JSON.stringify({ type: 'init', groups: parsed }));
                                fileInfo.textContent = `✅ 已上传: ${file.name} (${parsed.length} 个分组)`;
                                showToast('📤 数据已发送至服务器', 'success');
                                state.groups = parsed;
                            } else {
                                showToast('❌ 服务器未连接', 'error');
                            }
                        }
                    } catch (err) {
                        showToast('❌ 解析失败: ' + err.message, 'error');
                        console.error('解析错误:', err);
                    }
                };
                reader.readAsArrayBuffer(file);
            }

            function parseExcelData(rows) {
                let kCol = null,
                    asCol = null,
                    statusCol = null;
                const headers = Object.keys(rows[0] || {});
                for (const h of headers) {
                    const clean = h.trim();
                    if (clean.includes('FBA货件ID') || clean.includes('FBAShipmentID') || clean.includes('货件ID')) kCol = h;
                    if (clean.includes('Custom box barcode') || clean.includes('自定义箱条码') || clean.includes('箱条码')) asCol =
                    h;
                    if (clean.includes('Status') || clean.includes('状态')) statusCol = h;
                }
                if (!kCol && !asCol) {
                    showToast('未找到K列或AS列', 'error');
                    return null;
                }

                const raw = [];
                for (const row of rows) {
                    let kVal = kCol ? String(row[kCol] || '').trim() : '';
                    let asVal = asCol ? String(row[asCol] || '').trim() : '';
                    let statusVal = statusCol ? String(row[statusCol] || '').trim() : '';
                    let code = kVal || asVal;
                    if (!code) continue;
                    let type = 'fba',
                        prefix = '';
                    if (kVal) { type = 'fba';
                        prefix = kVal; } else if (asVal) { type = 'custom';
                        prefix = extractPrefix(asVal); } else continue;
                    raw.push({ k: kVal, as: asVal, code: asVal || kVal, prefix, type, status: statusVal });
                }

                const map = new Map();
                for (const item of raw) {
                    const key = item.prefix;
                    const gkey = key + '|' + item.type;
                    if (!map.has(gkey)) {
                        map.set(gkey, { key, type: item.type, items: [], hasCanceled: false });
                    }
                    const g = map.get(gkey);
                    g.items.push({ code: item.code, raw: item });
                    if (item.status === '已取消') g.hasCanceled = true;
                }
                const groups = [];
                for (const [_, g] of map) {
                    g.items.sort((a, b) => {
                        const na = parseInt(a.code.replace(/^.*?(\d+)$/, '$1'));
                        const nb = parseInt(b.code.replace(/^.*?(\d+)$/, '$1'));
                        if (!isNaN(na) && !isNaN(nb)) return na - nb;
                        return a.code.localeCompare(b.code);
                    });
                    groups.push(g);
                }
                groups.sort((a, b) => {
                    if (a.type !== b.type) return a.type === 'fba' ? -1 : 1;
                    return a.key.localeCompare(b.key);
                });
                return groups;
            }

            function extractPrefix(val) {
                if (!val) return val;
                const s = String(val).trim();
                if (s.includes('/')) return s.split('/')[0].trim();
                const m = s.match(/^(.*?[A-Za-z])U\d+$/);
                if (m) return m[1] + 'U';
                const m2 = s.match(/^(.*?[A-Za-z])\d+$/);
                if (m2 && m2[1].length >= 2) return m2[1];
                return s;
            }

            // ---- 导出 ----
            function exportTxt() {
                const lines = [];
                lines.push('出库复核清单');
                lines.push(`生成: ${new Date().toLocaleString()}`);
                lines.push('─'.repeat(50));
                for (const g of state.groups) {
                    const typeLabel = g.type === 'fba' ? 'FBA货件' : '自定义箱条码';
                    lines.push(`【${typeLabel}: ${g.key}】 ${g.items.length} 箱`);
                    for (let i = 0; i < g.items.length; i++) {
                        const item = g.items[i];
                        let displayCode = item.code;
                        if (g.type === 'fba') displayCode = `${g.key}/${i+1}`;
                        const scanned = state.scanStatus[item.code] || false;
                        lines.push(`  ${String(i+1).padStart(3,' ')}. ${scanned ? '✅' : '⬜'} ${displayCode}`);
                    }
                    lines.push('');
                }
                const blob = new Blob([lines.join('\n')], { type: 'text/plain;charset=utf-8' });
                const a = document.createElement('a');
                a.href = URL.createObjectURL(blob);
                a.download = `复核清单_${new Date().toISOString().slice(0,10)}.txt`;
                a.click();
                URL.revokeObjectURL(a.href);
                showToast('📄 TXT已导出', 'success');
            }

            function exportExcel() {
                const rows = [
                    ['#', '分组类型', '分组Key', '子项序号', '子项条码', '已扫描']
                ];
                let idx = 0;
                for (const g of state.groups) {
                    const typeLabel = g.type === 'fba' ? 'FBA货件' : '自定义箱条码';
                    idx++;
                    rows.push([idx, typeLabel, g.key, '', `共${g.items.length}箱`, '']);
                    for (let i = 0; i < g.items.length; i++) {
                        const item = g.items[i];
                        let displayCode = item.code;
                        if (g.type === 'fba') displayCode = `${g.key}/${i+1}`;
                        const scanned = state.scanStatus[item.code] || false;
                        rows.push(['', '', '', i + 1, displayCode, scanned ? '是' : '否']);
                    }
                }
                const ws = XLSX.utils.aoa_to_sheet(rows);
                const wb = XLSX.utils.book_new();
                XLSX.utils.book_append_sheet(wb, ws, '复核清单');
                const out = XLSX.write(wb, { bookType: 'xlsx', type: 'array' });
                const blob = new Blob([out], { type: 'application/octet-stream' });
                const a = document.createElement('a');
                a.href = URL.createObjectURL(blob);
                a.download = `复核清单_${new Date().toISOString().slice(0,10)}.xlsx`;
                a.click();
                URL.revokeObjectURL(a.href);
                showToast('📊 Excel已导出', 'success');
            }

                    scanInput.focus();
                    console.log('🔍 输入框已获得焦点');
                }, 500);
            }

            if (document.readyState === 'loading') {
                document.addEventListener('DOMContentLoaded', init);
            } else {
                init();
            }
        })();
